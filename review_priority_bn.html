§১

স্কোরিং নিয়ম

পাঁচটি স্বাধীন সংকেত। প্রতিটি শর্ত পূরণ হলে স্কোরে পয়েন্ট যোগ হয়। প্রতিটি threshold একটি করে command-line প্যারামিটার, তাই রান-কমান্ড থেকেই সব মান দেখা যায় ও বদলানো যায় — কোনোটিই কোডের ভেতরে লুকানো নেই।

প্রথম তিনটি আর শেষ দুটি সংকেত এক জিনিস মাপে না

প্রথম তিনটি মাপে কতটা হেডিং হয়েছে। শেষ দুটি মাপে পাইপলাইন নিজের আউটপুটকে কতটা কম বিশ্বাস করছে। ফলে বাড়তি একটিও হেডার ছাড়াই কেবল ডিটেক্টরের অনিশ্চয়তার কারণে একজন প্লেয়ার Moderate-এ উঠে যেতে পারে। কোডের কমেন্টেই উদ্দেশ্যটা লেখা আছে — "weak detection confidence adds uncertainty, not certainty"।

এজন্যই আউটপুটটিকে ম্যানুয়াল রিভিউয়ের সিরিয়াল হিসেবে দেখা উচিত ("কোন ক্লিপগুলো একজন মানুষের দেখা দরকার?"), এক্সপোজারের র‍্যাঙ্কিং হিসেবে নয় ("কে সবচেয়ে বেশি হেড করেছে?")। অনিশ্চিত ডেটা আর বেশি এক্সপোজার — দুটোই একটি ক্লিপ রিভিউয়ারের সামনে পাঠানোর বৈধ কারণ।

কোডে যেভাবে আছে

scripts/header_analysis/analyze_exposure.py

ডিফল্ট মানগুলো কমেন্ট হিসেবে পাশে বসিয়ে দেওয়া হলো।

def review_priority(total_headers, max_rolling_15min, min_interval_sec,
                    ambiguous_ratio, avg_header_confidence,
                    high_attrib_ratio, args) -> str:
    score = 0

    # ১. মোট এক্সপোজার
    if total_headers >= args.pr_total_high:              # 6
        score += 2
    elif total_headers >= args.pr_total_moderate:        # 3
        score += 1

    # ২. সময়ভিত্তিক গুচ্ছ
    if max_rolling_15min >= args.pr_cluster_high:        # 4
        score += 2
    elif max_rolling_15min >= args.pr_cluster_moderate:  # 2
        score += 1

    # ৩. দ্রুত পরপর হেডার (একটিমাত্র হেডার হলে NaN, নিয়মটি বাদ যায়)
    if not pd.isna(min_interval_sec) and \
       min_interval_sec <= args.pr_short_interval_sec:   # 60.0
        score += 1

    # ৪. ডিটেকশন অনিশ্চয়তা
    if avg_header_confidence < args.pr_low_confidence_cutoff:   # 0.6
        score += 1

    # ৫. অ্যাট্রিবিউশন অনিশ্চয়তা
    if ambiguous_ratio > args.pr_high_ambiguous_ratio:          # 0.3
        score += 1

    if score >= 4:
        return "High Review Priority"
    if score >= 2:
        return "Moderate Review Priority"
    return "Low Review Priority"

ইনপুট পাঁচটি রাশিই আসে summarize_players() থেকে, যেখানে প্রতিটি (video_id, track_id) জোড়ার নিশ্চিত হেডার ইভেন্টগুলো একসাথে করে total, max_rolling_15min, min_interval, গড় confidence আর ambiguous অনুপাত বের করা হয়।
